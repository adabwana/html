JavaScript Variables

Variables are containers for storing data values. JavaScript has three ways to declare variables.

Variable Declaration

// var (older way, avoid using)
var name = "John";

// let (modern way, can be reassigned)
let age = 25;
age = 26; // This is allowed

// const (cannot be reassigned)
const PI = 3.14159;
// PI = 3.14; // This would cause an error

Data Types

// Primitive data types
let string = "Hello World";    // String
let number = 42;               // Number
let boolean = true;            // Boolean
let nothing = null;            // Null
let undefined;                 // Undefined

// Complex data types
let array = [1, 2, 3, 4, 5];   // Array
let object = {name: "John", age: 25}; // Object

JavaScript Functions

Functions are reusable blocks of code that perform specific tasks.

Function Declaration

// Function declaration
function greetUser(name) {
    return "Hello, " + name + "!";
}

// Function expression
const addNumbers = function(a, b) {
    return a + b;
};

// Arrow function (ES6+)
const multiply = (a, b) => a * b;

// Calling functions
let greeting = greetUser("Alice");
let sum = addNumbers(5, 3);
let product = multiply(4, 7);

DOM Manipulation

// Selecting elements
let element = document.getElementById('myId');
let elements = document.querySelectorAll('.myClass');

// Changing content
element.textContent = "New text";
element.innerHTML = "<strong>Bold text</strong>";

// Changing styles
element.style.color = "red";
element.style.fontSize = "20px";

// Adding event listeners
element.addEventListener('click', function() {
    alert('Element clicked!');
});

Event Handling

// Click event
button.addEventListener('click', function() {
    console.log('Button was clicked!');
});

// Mouse events
element.addEventListener('mouseenter', function() {
    this.style.backgroundColor = 'lightblue';
});

element.addEventListener('mouseleave', function() {
    this.style.backgroundColor = '';
});

// Keyboard events
input.addEventListener('keydown', function(event) {
    console.log('Key pressed:', event.key);
});
